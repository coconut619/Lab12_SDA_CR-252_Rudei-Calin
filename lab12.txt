#include <stdio.h>
#include <stdlib.h>
#include <time.h>

long long comparatii = 0;
long long deplasari = 0;

void insertionSort(int a[], int n)
{
    int i, j, cheie;

    for(i = 1; i < n; i++)
    {
        cheie = a[i];
        j = i - 1;

        while(j >= 0)
        {
            comparatii++;

            if(a[j] > cheie)
            {
                a[j + 1] = a[j];
                deplasari++;
                j--;
            }
            else
                break;
        }

        a[j + 1] = cheie;
        deplasari++;
    }
}

void afisare(int a[], int n)
{
    int i;

    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
}

void aleator(int a[], int n)
{
    int i;

    for(i = 0; i < n; i++)
        a[i] = rand() % 10000;
}

void crescator(int a[], int n)
{
    int i;

    for(i = 0; i < n; i++)
        a[i] = i;
}

void descrescator(int a[], int n)
{
    int i;

    for(i = 0; i < n; i++)
        a[i] = n - i;
}

int cmp(const void *a, const void *b)
{
    return (*(int*)a - *(int*)b);
}

int main()
{
    srand(time(NULL));

    int opt;

    do
    {
        printf("\n===== MENIU =====\n");
        printf("1. Tablou demonstrativ\n");
        printf("2. Tablou aleator\n");
        printf("3. Tablou sortat crescator\n");
        printf("4. Tablou sortat descrescator\n");
        printf("5. Analiza qsort()\n");
        printf("0. Iesire\n");
        printf("Optiunea: ");
        scanf("%d", &opt);

        if(opt >= 1 && opt <= 5)
        {
            int n;

            if(opt == 1)
            {
                printf("n (5..20) = ");
                scanf("%d", &n);

                int a[20];

                printf("Introdu elementele:\n");
                for(int i = 0; i < n; i++)
                    scanf("%d", &a[i]);

                comparatii = 0;
                deplasari = 0;

                clock_t start = clock();

                insertionSort(a, n);

                clock_t end = clock();

                double timp =
                    (double)(end - start) / CLOCKS_PER_SEC;

                printf("\nTablou sortat:\n");
                afisare(a, n);

                printf("Comparatii = %lld\n", comparatii);
                printf("Deplasari = %lld\n", deplasari);
                printf("Timp = %f secunde\n", timp);
            }

            else if(opt >= 2 && opt <= 5)
            {
                int valori[3] = {10000, 100000, 1000000};

                for(int k = 0; k < 3; k++)
                {
                    n = valori[k];

                    int *a = (int*)malloc(n * sizeof(int));

                    if(opt == 2)
                        aleator(a, n);
                    else if(opt == 3)
                        crescator(a, n);
                    else if(opt == 4)
                        descrescator(a, n);

                    if(opt != 5)
                    {
                        comparatii = 0;
                        deplasari = 0;

                        clock_t start = clock();

                        insertionSort(a, n);

                        clock_t end = clock();

                        double timp =
                            (double)(end - start)
                            / CLOCKS_PER_SEC;

                        printf("\nn = %d\n", n);
                        printf("Comparatii = %lld\n",
                               comparatii);
                        printf("Deplasari = %lld\n",
                               deplasari);
                        printf("Timp = %f secunde\n",
                               timp);
                    }
                    else
                    {
                        aleator(a, n);

                        clock_t start = clock();

                        qsort(a, n, sizeof(int), cmp);

                        clock_t end = clock();

                        double timp =
                            (double)(end - start)
                            / CLOCKS_PER_SEC;

                        printf("\nqsort() pentru n = %d\n",
                               n);
                        printf("Timp = %f secunde\n",
                               timp);
                    }

                    free(a);
                }
            }
        }

    } while(opt != 0);

    return 0;
}
